$entries = @(
  @{ path='crates/cli/src/paths.rs'; line=57; name='canonical_root' },
  @{ path='crates/cli/src/paths.rs'; line=72; name='contains' },
  @{ path='crates/cli/src/main.rs'; line=184; name='resolve_recorded_root' },
  @{ path='crates/cli/src/json.rs'; line=7; name='escape' },
  @{ path='crates/cli/src/store.rs'; line=27; name='grammar_versions' }
)
foreach ($e in $entries) {
  $path = $e.path
  $bytes = [System.IO.File]::ReadAllBytes($path)
  $text = [System.Text.Encoding]::UTF8.GetString($bytes)
  $lineStarts = New-Object System.Collections.Generic.List[int]
  $lineStarts.Add(0)
  for ($i=0; $i -lt $text.Length; $i++) {
    if ($text[$i] -eq "`n") { $lineStarts.Add($i+1) }
  }
  $lineStart = $lineStarts[$e.line - 1]
  $idx = $text.IndexOf($e.name, $lineStart)
  $lineEnd = if ($e.line -lt $lineStarts.Count) { $lineStarts[$e.line] } else { $text.Length }
  if ($idx -lt 0 -or $idx -ge $lineEnd) { Write-Output "NOTFOUND $($e.name)" ; continue }
  $startByte = [System.Text.Encoding]::UTF8.GetByteCount($text.Substring(0, $idx))
  $endByte = $startByte + [System.Text.Encoding]::UTF8.GetByteCount($e.name)
  Write-Output ("{0} {1}:{2} bytes {3}-{4}" -f $e.name, $path, $e.line, $startByte, $endByte)
}
