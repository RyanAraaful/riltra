$files = @(
  @{ path = 'crates/cli/src/paths.rs'; name = 'canonical_root'; line = 57 },
  @{ path = 'crates/cli/src/paths.rs'; name = 'contains'; line = 72 },
  @{ path = 'crates/cli/src/main.rs'; name = 'resolve_recorded_root'; line = 184 },
  @{ path = 'crates/cli/src/json.rs'; name = 'escape'; line = 7 },
  @{ path = 'crates/cli/src/store.rs'; name = 'grammar_versions'; line = 27 }
)
foreach ($f in $files) {
  $bytes = [System.IO.File]::ReadAllBytes((Join-Path (Get-Location) $f.path))
  $text = [System.Text.Encoding]::UTF8.GetString($bytes)
  $lines = $text -split "`n"
  $lineText = $lines[$f.line - 1]
  $idxInLine = $lineText.IndexOf($f.name)
  $lineStartByte = 0
  for ($i = 0; $i -lt ($f.line - 1); $i++) {
    $lineStartByte += [System.Text.Encoding]::UTF8.GetByteCount($lines[$i] + "`n")
  }
  $prefixBytes = [System.Text.Encoding]::UTF8.GetByteCount($lineText.Substring(0, $idxInLine))
  $start = $lineStartByte + $prefixBytes
  $end = $start + [System.Text.Encoding]::UTF8.GetByteCount($f.name)
  Write-Output ("{0} {1}:{2} bytes {3}-{4}" -f $f.name, $f.path, $f.line, $start, $end)
}

$files = @(
  @{ path = 'crates/cli/src/paths.rs'; name = 'canonical_root'; start = 1704; end = 1718 },
  @{ path = 'crates/cli/src/paths.rs'; name = 'contains'; start = 2180; end = 2188 },
  @{ path = 'crates/cli/src/main.rs'; name = 'resolve_recorded_root'; start = 8095; end = 8116 },
  @{ path = 'crates/cli/src/json.rs'; name = 'escape'; start = 356; end = 362 },
  @{ path = 'crates/cli/src/store.rs'; name = 'grammar_versions'; start = 1289; end = 1305 }
)
foreach ($f in $files) {
  $bytes = [System.IO.File]::ReadAllBytes((Join-Path (Get-Location) $f.path))
  $slice = [System.Text.Encoding]::UTF8.GetString($bytes[$f.start..($f.end-1)])
  $ok = ($slice -eq $f.name)
  Write-Output ("{0} expected={1} got='{2}' ok={3}" -f $f.name, $f.name, $slice, $ok)
}
